      rok::core::iter

      Iterator + IntoIterator + FromIterator (stdlib spec 2.7).

      for pat in xs desugars to let it = xs.into_iter(); loop { match it.next() { v: T => { let pat = v; body } none => break } } (integer ranges and arrays keep a direct counting loop).

      next takes ref self (2.7.1): an iterator's state lives in its fields, written back into the caller's binding after each call.

/// The items multiplied; `1` for none.
const fn product(self) -> Item where Item: Mul<Output = Item> + One {
    let acc = Item::one();
    for x in self {
        acc *= x;
    }
    acc
}

const fn fold<B>(self, init: B, f: fn(B, Item) -> B) -> B {
    let acc = init;
    for x in self {
        acc = f(acc, x);
    }
    acc
}

/// The items combined left to right with `f`; `none` for none.
const fn reduce(self, f: fn(Item, Item) -> Item) -> Item | none {
    let acc: Item | none = none;
    for x in self {
        acc = if let a: Item = acc { f(a, x) } else { x };
    }
    acc
}

const fn any(self, pred: fn(Item) -> bool) -> bool {
    for x in self {
        if pred(x) {
            return true;
        }
    }
    false
}

const fn all(self, pred: fn(Item) -> bool) -> bool {
    for x in self {
        if !pred(x) {
            return false;
        }
    }
    true
}

const fn find(self, pred: fn(Item) -> bool) -> Item | none {
    for x in self {
        if pred(x) {
            return x;
        }
    }
    none
}

const fn position(self, pred: fn(Item) -> bool) -> usize | none {
    let i: usize = 0;
    for x in self {
        if pred(x) {
            return i;
        }
        i += 1;
    }
    none
}

const fn min(self) -> Item | none where Item: Ord {
    let best: Item | none = none;
    for x in self {
        match best {
            b: Item => {
                if x.cmp(b).is_lt() {
                    best = x;
                }
            }
            none => {
                best = x;
            }
        }
    }
    best
}

const fn max(self) -> Item | none where Item: Ord {
    let best: Item | none = none;
    for x in self {
        match best {
            b: Item => {
                if x.cmp(b).is_gt() {
                    best = x;
                }
            }
            none => {
                best = x;
            }
        }
    }
    best
}

/// The item with the smallest `key` (the first of equals).
const fn min_by_key<K: Ord>(self, key: fn(Item) -> K) -> Item | none {
    let best: Item | none = none;
    let best_key: K | none = none;
    for x in self {
        let k = key(x);
        match best_key {
            b: K => {
                if k.cmp(b).is_lt() {
                    best = x;
                    best_key = k;
                }
            }
            none => {
                best = x;
                best_key = k;
            }
        }
    }
    best
}

/// The item with the largest `key` (the last of equals).
const fn max_by_key<K: Ord>(self, key: fn(Item) -> K) -> Item | none {
    let best: Item | none = none;
    let best_key: K | none = none;
    for x in self {
        let k = key(x);
        match best_key {
            b: K => {
                if !k.cmp(b).is_lt() {
                    best = x;
                    best_key = k;
                }
            }
            none => {
                best = x;
                best_key = k;
            }
        }
    }
    best
}

const fn last(self) -> Item | none {
    let last: Item | none = none;
    for x in self {
        last = x;
    }
    last
}

/// The `n`th remaining item (0-based), consuming the ones before it.
const fn nth(ref self, n: usize) -> Item | none {
    for i in 0..n {
        self.next()?;
    }
    self.next()
}

const fn for_each(self, f: fn(Item)) {
    for x in self {
        f(x);
    }
}

/// Gather the items into a `FromIterator` collection (the slot's
/// type picks it: `let v: List<i32> = it.collect();`).
const fn collect<C: FromIterator<Item>>(self) -> C {
    C::from_iter(self)
}
